# Atlas Engine — Phase 0 Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `atlas_engine`, a Python package that records a browser session, deterministically extracts a step trajectory from it, replays it with a bounded LLM-repair loop ("the brain"), emits a self-contained Playwright script, and certifies it — proving the core "record → certified golden script" loop end to end against a local fixture portal, with zero LLM credentials required for the automated test suite.

**Architecture:** One Python package (no TypeScript, no generated-code-as-strings) implementing capture/trajectory/brain/codegen/verify natively, per `docs/superpowers/specs/2026-08-12-atlas-ai-production-architecture-design.md` §6. Postgres (via SQLAlchemy + Alembic) stores metadata per `docs/superpowers/specs/2026-08-12-atlas-ai-database-design.md` and `db/schema.sql`; recording/script artifacts live on local disk under a run directory, referenced by path (the local stand-in for the Blob Storage refs the cloud platform will use later). A CLI (`atlas record` / `atlas author`) and a thin local FastAPI service both drive the same engine calls, so this is ready to be wired to a desktop shell in the next plan.

**Tech Stack:** Python 3.11+ (developed/tested on 3.14.6), Playwright for Python, FastAPI + Uvicorn, SQLAlchemy 2.0 + Alembic, psycopg3, Pydantic v2, Click, pytest, Postgres 16 via Docker Compose for local dev.

## Global Constraints

- No LLM call ever ships inside an emitted golden script — the emitted script is plain `playwright.sync_api` code with zero dependency on `atlas_engine` or any network call to an LLM. (Architecture spec, Guiding Principle 2.)
- The recording is never distilled or regenerated by an LLM — every stage downstream of capture operates on the literal recorded trajectory. (Guiding Principle 1.)
- Every table in the data model carries `org_id`, cascading from `organizations`, even though this plan only ever creates one organization. (Database design doc, "org_id everywhere.")
- No manual JSON config authoring for a portal — `portal_profiles` rows are inferred from a recording, never hand-written. (Architecture spec §6.3.)
- No automated test in this plan may require a real Azure OpenAI key, a real healthcare portal, or a real human clicking anything — the LLM client is dependency-injected so a fake stands in for tests, and the fixture portal stands in for a real site.
- Selectors are ranked by durability: `data-testid` / `id` &gt; ARIA role + accessible name &gt; CSS &gt; text &gt; XPath — carried forward from the prototype's locator design (architecture spec §16, "ported deliberately").
- Every step ends with tests passing and a commit. Do not proceed to the next task with failing tests.

```python
from contextlib import contextmanager

from sqlalchemy import create_engine
from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker

from atlas_engine.config import settings


class Base(DeclarativeBase):
    pass


_engine = None
_SessionFactory = None


def get_engine(database_url: str | None = None):
    global _engine
    if _engine is None:
        _engine = create_engine(database_url or settings.database_url, future=True)
    return _engine


def get_session_factory(database_url: str | None = None) -> sessionmaker:
    global _SessionFactory
    if _SessionFactory is None:
        _SessionFactory = sessionmaker(bind=get_engine(database_url), expire_on_commit=False)
    return _SessionFactory


@contextmanager
def session_scope(database_url: str | None = None):
    session: Session = get_session_factory(database_url)()
    try:
        yield session
        session.commit()
    except Exception:
        session.rollback()
        raise
    finally:
        session.close()
```

from atlas_engine.db import Base

ORG_PLANS = ("internal", "team", "enterprise")
USER_ROLES = ("owner", "admin", "member")


class Organization(Base):
    __tablename__ = "organizations"

    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    name: Mapped[str]
    slug: Mapped[str] = mapped_column(unique=True)
    plan: Mapped[str] = mapped_column(Enum(*ORG_PLANS, name="org_plan"), default="internal")
    settings: Mapped[dict] = mapped_column(JSONB, default=dict)
    created_at: Mapped[datetime] = mapped_column(server_default=func.now())
    updated_at: Mapped[datetime] = mapped_column(server_default=func.now(), onupdate=func.now())

    users: Mapped[list["User"]] = relationship(back_populates="org", cascade="all, delete-orphan")


class User(Base):
    __tablename__ = "users"
    __table_args__ = (UniqueConstraint("org_id", "email"),)

```python
import json
import uuid
from dataclasses import dataclass
from pathlib import Path
from typing import Any


@dataclass
class RunDir:
    path: Path
    blob_ref: str

    def write_json(self, name: str, data: Any) -> str:
        (self.path / name).write_text(json.dumps(data, indent=2))
        return f"{self.blob_ref}/{name}"

    def read_json(self, name: str) -> Any:
        return json.loads((self.path / name).read_text())

    def write_bytes(self, name: str, data: bytes) -> str:
        (self.path / name).write_bytes(data)
        return f"{self.blob_ref}/{name}"


class LocalRunStore:
    """On-disk stand-in for the cloud Blob Storage backend.

    A `RunDir`'s `blob_ref` is a path relative to `root` — the same shape
    a Blob Storage key will take later, so callers never need to know
    whether they're talking to local disk or Blob Storage.
    """

    def __init__(self, root: Path):
        self.root = Path(root)
        self.root.mkdir(parents=True, exist_ok=True)

    def new_run_dir(self, prefix: str) -> RunDir:
        run_id = str(uuid.uuid4())
        blob_ref = f"{prefix}/{run_id}"
        path = self.root / blob_ref
        path.mkdir(parents=True, exist_ok=True)
        return RunDir(path=path, blob_ref=blob_ref)
```

import re

_TAG_TEXT_RE = re.compile(r">([^<>]+)<", re.MULTILINE)


def sanitize_dom(html: str) -> str:
    """Strip inter-tag text content while keeping tags, attributes, and
    structure intact — an LLM diagnosing a broken selector needs the shape
    of the page, never the PHI-bearing values rendered on it."""

    def _blank(match: re.Match) -> str:
        text = match.group(1)
        if text.strip() == "":
            return match.group(0)
        return ">" + " " * len(text) + "<"

    return _TAG_TEXT_RE.sub(_blank, html)


def sanitize_text_value(value: str) -> str:
    if value == "":
        return ""
    return "<REDACTED>"
```

import json
from typing import Protocol

import httpx


class LlmClient(Protocol):
    def complete(
        self,
        prompt: str,
        *,
        images: list[bytes] | None = None,
        json_mode: bool = True,
        temperature: float = 0.0,
    ) -> dict: ...


class FakeLlmClient:
    """Test double — queue canned responses, record every call made."""

    def __init__(self, responses: list[dict]):
        self._responses = list(responses)
        self.calls: list[dict] = []

    def complete(
        self,
        prompt: str,
        *,
        images: list[bytes] | None = None,
        json_mode: bool = True,
        temperature: float = 0.0,
    ) -> dict:
        self.calls.append(
            {"prompt": prompt, "images": images, "json_mode": json_mode, "temperature": temperature}
        )
        return self._responses.pop(0)


class AzureOpenAiClient:
    """Real client — used in production, never exercised by this plan's
    automated tests (no credentials required to run the test suite)."""

    def __init__(self, endpoint: str, api_key: str, deployment: str, api_version: str = "2024-10-21"):
        self.endpoint = endpoint.rstrip("/")
        self.api_key = api_key
        self.deployment = deployment
        self.api_version = api_version

    def complete(
        self,
        prompt: str,
        *,
        images: list[bytes] | None = None,
        json_mode: bool = True,
        temperature: float = 0.0,
    ) -> dict:
        content: list[dict] = [{"type": "text", "text": prompt}]
        for image_bytes in images or []:
            import base64

            b64 = base64.b64encode(image_bytes).decode("ascii")
            content.append({"type": "image_url", "image_url": {"url": f"data:image/png;base64,{b64}"}})

        url = (
            f"{self.endpoint}/openai/deployments/{self.deployment}/chat/completions"
            f"?api-version={self.api_version}"
        )
        payload = {
            "messages": [{"role": "user", "content": content}],
            "temperature": temperature,
        }
        if json_mode:
            payload["response_format"] = {"type": "json_object"}

        response = httpx.post(url, headers={"api-key": self.api_key}, json=payload, timeout=60.0)
        response.raise_for_status()
        message = response.json()["choices"][0]["message"]["content"]
        return json.loads(message)
```

```python
from atlas_engine.capture.events import RecordedAction

_LISTENER_SCRIPT = """
() => {
  window.__atlasEvents = [];
  const push = (kind, el, value) => {
    const rect = el.getBoundingClientRect
      ? el.getBoundingClientRect()
      : { x: 0, y: 0 };
    window.__atlasEvents.push({
      kind,
      value: value ?? null,
      tag: el.tagName || null,
      id: el.id || null,
      testId: el.getAttribute ? el.getAttribute("data-testid") : null,
      x: rect.x,
      y: rect.y,
    });
  };
  document.addEventListener(
    "click",
    (event) => push("click", event.target, null),
    true,
  );
  document.addEventListener(
    "input",
    (event) => push("fill", event.target, event.target.value),
    true,
  );
}
"""


def _selector_candidates(record: dict) -> list[dict]:
    candidates = []
    if record.get("testId"):
        candidates.append({"selector": f'[data-testid="{record["testId"]}"]', "rank": 0})
    if record.get("id"):
        candidates.append({"selector": f'#{record["id"]}', "rank": 1})
    return candidates


class CaptureSession:
    """Records clicks and fills on a live Page using only public Playwright
    APIs (page.on, add_init_script) — never the private enableRecorder
    channel the prototype relied on."""

    def __init__(self, page):
        self.page = page
        self.actions: list[RecordedAction] = []
        self._drained = 0

    def __enter__(self) -> "CaptureSession":
        self.page.evaluate(_LISTENER_SCRIPT)
        self.page.on("framenavigated", self._on_navigate)
        return self

    def __exit__(self, exc_type, exc_val, exc_tb) -> None:
        self._drain()

    def _drain(self) -> None:
        dom_after = self.page.content()
        url_after = self.page.url
        records = self.page.evaluate("window.__atlasEvents || []")
        for record in records[self._drained :]:
            self.actions.append(
                RecordedAction(
                    index=len(self.actions),
                    kind=record["kind"],
                    selector_candidates=_selector_candidates(record),
                    value=record["value"],
                    url_before=self.actions[-1].url_after if self.actions else url_after,
                    url_after=url_after,
                    dom_before=self.actions[-1].dom_after if self.actions else dom_after,
                    dom_after=dom_after,
                )
            )
        self._drained = len(records)

    def _on_navigate(self, frame) -> None:
        if frame == self.page.main_frame:
            self._drain()
```

from atlas_engine.capture.events import RecordedAction


def _expected_effect(action: RecordedAction) -> str:
    if action.url_before != action.url_after:
        return "url_changed"
    if action.dom_before != action.dom_after:
        return "dom_changed"
    return "none"


def extract_trajectory(actions: list[RecordedAction]) -> ReferenceTrajectory:
    """Deterministic, no LLM — a pure function of the recorded actions.

    steps = tuple(

            kind=action.kind,
            selector_candidates=action.selector_candidates,
            value=action.value,
            expected_effect=_expected_effect(action),
        )
        for action in actions
    )
    return ReferenceTrajectory(steps=steps)

from dataclasses import dataclass

from atlas_engine.brain.models import TrajectoryStep


@dataclass(frozen=True)
class StepOutcome:
    step_index: int
    on_track: bool
    url_after: str
    dom_after: str


def evaluate(
    step: TrajectoryStep,
    *,
    url_before: str,
    url_after: str,
    dom_before: str,
    dom_after: str,
) -> StepOutcome:
    """Deterministic — compares the step's expected effect (recorded at
    capture time) against what actually happened live. This is the check
    that decides whether the brain needs to think at all."""
    if step.expected_effect == "url_changed":
        on_track = url_before != url_after
    elif step.expected_effect == "dom_changed":
        on_track = dom_before != dom_after
    else:
        on_track = True

    return StepOutcome(step_index=step.index, on_track=on_track, url_after=url_after, dom_after=dom_after)

from collections.abc import Callable

from atlas_engine.brain.evaluate import StepOutcome, evaluate
from atlas_engine.brain.models import ReferenceTrajectory, TrajectoryStep


def _first_selector(step: TrajectoryStep) -> str:
    ranked = sorted(step.selector_candidates, key=lambda c: c["rank"])
    return ranked[0]["selector"]


def run_step(page, step: TrajectoryStep) -> StepOutcome:
    url_before = page.url
    dom_before = page.content()

    selector = _first_selector(step)
    if step.kind == "click":
        page.click(selector)
    elif step.kind == "fill":
        page.fill(selector, step.value or "")
    else:
        raise ValueError(f"Unsupported step kind: {step.kind}")

```python
from dataclasses import dataclass
from typing import Literal

from atlas_engine.brain.evaluate import StepOutcome
from atlas_engine.brain.models import TrajectoryStep
from atlas_engine.llm.client import LlmClient
from atlas_engine.llm.sanitize import sanitize_dom

_VALID_ACTIONS = {"replace_selector", "force_click", "wait_longer", "give_up"}


@dataclass(frozen=True)
class RepairAction:
    action: Literal["replace_selector", "force_click", "wait_longer", "give_up"]
    selector: str | None = None


def _build_prompt(step: TrajectoryStep, sanitized_dom: str) -> str:
    original_selector = step.selector_candidates[0]["selector"] if step.selector_candidates else None
    return (
        "A recorded Playwright step is off-track. The step was a "
        f"'{step.kind}' using selector '{original_selector}', expecting effect "
        f"'{step.expected_effect}'. Here is the current, sanitized page DOM "
        f"(text content stripped, structure and attributes intact):\n\n{sanitized_dom}\n\n"
        "Respond in JSON with one of: "
        '{"action": "replace_selector", "selector": "<css selector present in the DOM above>"}, '
        '{"action": "force_click"}, {"action": "wait_longer"}, or {"action": "give_up"}.'
    )


def think(llm: LlmClient, step: TrajectoryStep, page, outcome: StepOutcome) -> RepairAction:
    """Bounded, DOM-validated repair proposal — the only place in the
    engine an LLM call happens. Any proposed selector that doesn't
    actually exist on the live page is coerced to give_up rather than
    trusted blindly."""
    sanitized_dom = sanitize_dom(page.content())
    prompt = _build_prompt(step, sanitized_dom)
    response = llm.complete(prompt, json_mode=True, temperature=0.0)

    action = response.get("action")
    if action not in _VALID_ACTIONS:
        return RepairAction(action="give_up")

    if action == "replace_selector":
        selector = response.get("selector")
        if not selector or page.locator(selector).count() == 0:
            return RepairAction(action="give_up")
        return RepairAction(action="replace_selector", selector=selector)

    return RepairAction(action=action)
```

```python
from collections.abc import Callable

from atlas_engine.brain.evaluate import StepOutcome, evaluate
from atlas_engine.brain.models import ReferenceTrajectory, TrajectoryStep
from atlas_engine.brain.repair import think
from atlas_engine.llm.client import LlmClient


def _first_selector(step: TrajectoryStep) -> str:
    ranked = sorted(step.selector_candidates, key=lambda c: c["rank"])
    return ranked[0]["selector"]


def _act(page, selector: str, step: TrajectoryStep, *, force: bool = False) -> None:
    if step.kind == "click":
        page.click(selector, force=force)
    elif step.kind == "fill":
        page.fill(selector, step.value or "")
    else:
        raise ValueError(f"Unsupported step kind: {step.kind}")


def run_step(page, step: TrajectoryStep, *, selector: str | None = None, force: bool = False) -> StepOutcome:
    url_before = page.url
    dom_before = page.content()

    _act(page, selector or _first_selector(step), step, force=force)

    page.wait_for_load_state("domcontentloaded")
    url_after = page.url
    dom_after = page.content()

    return evaluate(step, url_before=url_before, url_after=url_after, dom_before=dom_before, dom_after=dom_after)


class BrainLoop:
    """perceive -> act -> evaluate -> think only if off-track.

    Without `llm`, an off-track step is reported via `on_off_track` and
    left off-track (Task 8 behavior, still used by tests that only exercise
    the happy path). With `llm`, an off-track step triggers exactly one
    bounded repair attempt via `think()` before moving on.
    """

    def __init__(
        self,
        page,
        on_off_track: Callable[[TrajectoryStep, StepOutcome], None] | None = None,
        llm: LlmClient | None = None,
    ):
        self.page = page
        self.on_off_track = on_off_track
        self.llm = llm

    def run(self, trajectory: ReferenceTrajectory) -> list[StepOutcome]:
        outcomes = []
        for step in trajectory.steps:
            outcome = run_step(self.page, step)

            if not outcome.on_track:
                if self.on_off_track is not None:
                    self.on_off_track(step, outcome)
                if self.llm is not None:
                    outcome = self._repair(step, outcome)

            outcomes.append(outcome)
        return outcomes

    def _repair(self, step: TrajectoryStep, outcome: StepOutcome) -> StepOutcome:
        repair = think(self.llm, step, self.page, outcome)
        if repair.action == "replace_selector":
            return run_step(self.page, step, selector=repair.selector)
        if repair.action == "force_click":
            return run_step(self.page, step, force=True)
        return outcome  # wait_longer / give_up: nothing more Task 9 attempts automatically
```

from atlas_engine.brain.models import ReferenceTrajectory, TrajectoryStep

_HEADER = '''"""Golden script — generated by Atlas AI. Deterministic replay only:
no LLM calls, no dependency on atlas_engine at runtime."""

import sys

from playwright.sync_api import sync_playwright


def run(start_url: str) -> None:
    with sync_playwright() as playwright:
        browser = playwright.chromium.launch(headless=True)
        page = browser.new_page()
        page.goto(start_url)

'''

_FOOTER = '''
        browser.close()


if __name__ == "__main__":
    run(sys.argv[1] if len(sys.argv) > 1 else {start_url!r})
'''


def _selector(step: TrajectoryStep) -> str:
    return sorted(step.selector_candidates, key=lambda c: c["rank"])[0]["selector"]


def _emit_step(step: TrajectoryStep) -> str:
    selector = _selector(step)
    if step.kind == "click":
        line = f'page.click({selector!r})'
    elif step.kind == "fill":
        line = f'page.fill({selector!r}, {step.value!r})'
    else:
        raise ValueError(f"Unsupported step kind for emission: {step.kind}")

    if step.expected_effect == "url_changed":
        line += f'\n        page.wait_for_load_state("domcontentloaded")'
    return f"        {line}"


def emit_golden_script(trajectory: ReferenceTrajectory, *, start_url: str) -> str:
    """Deterministic, no LLM — converts a verified trajectory into a
    self-contained Playwright script with zero atlas_engine dependency."""
    body = "\n".join(_emit_step(step) for step in trajectory.steps)
    return _HEADER + body + "\n" + _FOOTER.format(start_url=start_url)

import subprocess
import sys
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class CertificationResult:
    runs: int
    succeeded: int

    @property
    def is_certified(self) -> bool:
        return self.runs > 0 and self.succeeded == self.runs


def certify_golden_script(script_path: Path, *, args: list[str] | None = None, runs: int = 2) -> CertificationResult:
    """Re-runs a golden script K times from a fresh process each time to
    catch flakiness before it's marked certified."""
    succeeded = 0
    for _ in range(runs):
        result = subprocess.run(
            [sys.executable, str(script_path), *(args or [])],
            capture_output=True,
            text=True,
            timeout=60,
        )
        if result.returncode == 0:
            succeeded += 1
    return CertificationResult(runs=runs, succeeded=succeeded)

```python
from dataclasses import dataclass
from urllib.parse import urlparse

from atlas_engine.capture.events import RecordedAction


@dataclass(frozen=True)
class InferredProfile:
    domain: str
    login_url: str | None
    success_signal: dict | None
    confidence: float


def _is_password_action(action: RecordedAction) -> bool:
    return any("password" in c["selector"].lower() for c in action.selector_candidates)


def infer_portal_profile(start_url: str, actions: list[RecordedAction]) -> InferredProfile:
    domain = urlparse(start_url).netloc

    login_action = next((a for a in actions if _is_password_action(a)), None)
    login_url = login_action.url_before if login_action else start_url

    success_signal = None
    if login_action is not None:
        later_actions = [a for a in actions if a.index > login_action.index]
        navigated = next((a for a in later_actions if a.url_before != a.url_after), None)
        if navigated is not None:
            success_signal = {"url_contains": urlparse(navigated.url_after).path}

    if login_action is not None and success_signal is not None:
        confidence = 1.0
    elif login_action is not None or success_signal is not None:
        confidence = 0.5
    else:
        confidence = 0.5 if login_url else 0.0

    return InferredProfile(domain=domain, login_url=login_url, success_signal=success_signal, confidence=confidence)
```

@click.group()
def main() -> None:
    """Atlas AI engine CLI."""


@main.command("record")
@click.argument("url")
@click.option("--org-id", required=True)
def record(url: str, org_id: str) -> None:
    """Open a headed browser at URL, capture actions until you press Enter,
    persist the recording, and print its ID."""
    store = LocalRunStore(root=settings.ensure_run_root())
    run_dir = store.new_run_dir(prefix="recordings")

    with sync_playwright() as playwright:
        browser = playwright.chromium.launch(headless=False)
        page = browser.new_page()
        page.goto(url)

        with CaptureSession(page) as capture:
            click.echo("Recording... perform your steps in the opened browser, then press Enter here to finish.")
            input()

        browser.close()

    run_dir.write_json("actions.json", [vars(a) for a in capture.actions])

    with session_scope() as session:
        profile_data = infer_portal_profile(url, capture.actions)
        profile = PortalProfile(
            org_id=uuid.UUID(org_id),
            domain=profile_data.domain,
            inferred_login_url=profile_data.login_url,
            inferred_success_signal=profile_data.success_signal,
            confidence_score=profile_data.confidence,
        )
        session.add(profile)
        session.flush()

        recording = Recording(
            org_id=uuid.UUID(org_id),
            portal_profile_id=profile.id,
            status="captured",
            capture_blob_prefix=run_dir.blob_ref,
        )
        session.add(recording)
        session.flush()
        click.echo(f"Recording saved: {recording.id}")


def _author(org_id: str, start_url: str, actions: list[RecordedAction]) -> None:
    store = LocalRunStore(root=settings.ensure_run_root())
    run_dir = store.new_run_dir(prefix="goldens")

    trajectory = extract_trajectory(actions)
    script_text = emit_golden_script(trajectory, start_url=start_url)
    script_path = run_dir.path / "golden.py"
    script_path.write_text(script_text)

    certification = certify_golden_script(script_path, runs=2)

    with session_scope() as session:
        profile = PortalProfile(org_id=uuid.UUID(org_id), domain=start_url.split("/")[2])
        session.add(profile)
        session.flush()

        golden = GoldenScript(
            org_id=uuid.UUID(org_id),
            portal_profile_id=profile.id,
            version=1,
            script_blob_ref=str(script_path),
            status="certified" if certification.is_certified else "draft",
        )
        session.add(golden)
        session.flush()

    if certification.is_certified:
        click.echo(f"CERTIFIED golden script ({certification.succeeded}/{certification.runs} runs): {script_path}")
    else:
        click.echo(f"NOT certified ({certification.succeeded}/{certification.runs} runs): {script_path}")


@main.command("author-from-actions")
@click.option("--org-id", required=True)
@click.option("--start-url", required=True)
@click.option("--actions-file", required=True, type=click.Path(exists=True))
def author_from_actions(org_id: str, start_url: str, actions_file: str) -> None:
    """Author a golden script from a previously captured actions.json file
    (used by `atlas record`'s output, and directly by tests)."""
    raw = json.loads(Path(actions_file).read_text())
    actions = [RecordedAction(**item) for item in raw]
    _author(org_id, start_url, actions)


if __name__ == "__main__":
    main()
